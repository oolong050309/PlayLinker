### 2\. Pinia 状态管理 (`src/stores/`)

文档中提到了 `src/stores/price.js`，我们需要实现它来管理跨页面的价格监控状态。

#### 2.1 `src/stores/price.js`

```javascript
import { defineStore } from 'pinia'
import { ref, computed } from 'vue'
import { wishlistApi } from '@/api/wishlist'
import { priceApi } from '@/api/price'

export const usePriceStore = defineStore('price', () => {
  // State
  const wishlist = ref([])
  const alerts = ref([])
  const loading = ref(false)
  
  // Getters
  const totalPotentialSavings = computed(() => {
    // 假设后端没有直接返回总额，前端计算示例
    return wishlist.value.reduce((acc, item) => {
      if (item.currentPrice < item.originalPrice) {
        return acc + (item.originalPrice - item.currentPrice)
      }
      return acc
    }, 0)
  })

  const activeAlertsCount = computed(() => alerts.value.filter(a => a.isActive).length)

  // Actions
  const fetchWishlist = async () => {
    loading.value = true
    try {
      const res = await wishlistApi.getWishlist()
      if (res.success) {
        wishlist.value = res.data.items
      }
    } finally {
      loading.value = false
    }
  }

  const fetchAlerts = async () => {
    const res = await priceApi.getSubscriptions()
    if (res.success) {
      alerts.value = res.data.items
    }
  }

  return {
    wishlist,
    alerts,
    loading,
    totalPotentialSavings,
    activeAlertsCount,
    fetchWishlist,
    fetchAlerts
  }
})
```

<script setup>
import { ref, onMounted, nextTick } from 'vue'
import Chart from 'chart.js/auto'
import { usePriceStore } from '@/stores/price'
import { priceApi } from '@/api/price'
import { wishlistApi } from '@/api/wishlist'
import { createIcons, icons } from 'lucide'
import { ElMessage } from 'element-plus'

const store = usePriceStore()
const selectedGame = ref(null)
const chartCanvas = ref(null)
let chartInstance = null

const calculateDiscount = (game) => {
  if (!game.originalPrice) return 0
  return Math.round(((game.originalPrice - game.currentPrice) / game.originalPrice) * 100)
}

const selectGame = async (game) => {
  selectedGame.value = { ...game } // 浅拷贝避免直接修改Store
  const res = await priceApi.getPriceHistory(game.gameId)
  if (res.success) {
    nextTick(() => updateChart(res.data.priceHistory))
  }
}

const updateChart = (history) => {
  if (!chartCanvas.value) return
  if (chartInstance) chartInstance.destroy()

  const ctx = chartCanvas.value.getContext('2d')
  const gradient = ctx.createLinearGradient(0, 0, 0, 200)
  gradient.addColorStop(0, 'rgba(99, 102, 241, 0.2)')
  gradient.addColorStop(1, 'rgba(99, 102, 241, 0)')

  chartInstance = new Chart(ctx, {
    type: 'line',
    data: {
      labels: history.map(h => h.date), // 简化日期显示
      datasets: [{
        label: '价格',
        data: history.map(h => h.currentPrice),
        borderColor: '#6366f1',
        backgroundColor: gradient,
        fill: true,
        tension: 0.4,
        pointBackgroundColor: '#6366f1',
        pointRadius: 3
      }]
    },
    options: {
      responsive: true,
      maintainAspectRatio: false,
      plugins: { legend: { display: false } },
      scales: {
        y: { grid: { color: '#27272a' }, ticks: { color: '#71717a' } },
        x: { grid: { display: false }, ticks: { display: false } }
      }
    }
  })
}

const saveSettings = async () => {
  try {
    await wishlistApi.updateWishlist(selectedGame.value.subscriptionId, {
      targetPrice: selectedGame.value.targetPrice
    })
    ElMessage.success('监控设置已更新')
    store.fetchWishlist() // 刷新列表
  } catch (e) {
    ElMessage.error('保存失败')
  }
}

onMounted(async () => {
  await store.fetchWishlist()
  if (store.wishlist.length > 0) {
    selectGame(store.wishlist[0])
  }
  nextTick(() => createIcons({ icons }))
})
</script>
